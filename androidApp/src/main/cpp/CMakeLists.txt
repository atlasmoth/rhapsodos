cmake_minimum_required(VERSION 3.22.1)
project(rhapsodos_speech LANGUAGES C CXX)

add_library(rhapsodos_phonon SHARED PhononBridge.c)
target_include_directories(rhapsodos_phonon PRIVATE ../../../../native/phonon/include)
target_compile_options(rhapsodos_phonon PRIVATE -Wall -Wextra -Werror)
target_link_options(rhapsodos_phonon PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(rhapsodos_phonon PRIVATE dl)

add_library(whisper_runtime SHARED IMPORTED)
set_target_properties(whisper_runtime PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/../jniLibs/${ANDROID_ABI}/libwhisper.so")
add_library(ggml_base SHARED IMPORTED)
set_target_properties(ggml_base PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/../jniLibs/${ANDROID_ABI}/libggml-base.so")
add_library(rhapsodos_whisper SHARED WhisperBridge.cpp)
target_include_directories(rhapsodos_whisper PRIVATE ../../../../native/whisper/include)
target_compile_features(rhapsodos_whisper PRIVATE cxx_std_17)
target_compile_options(rhapsodos_whisper PRIVATE -Wall -Wextra -Werror)
target_link_options(rhapsodos_whisper PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(rhapsodos_whisper PRIVATE whisper_runtime ggml_base)
